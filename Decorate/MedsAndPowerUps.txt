			"####" A 0 A_PlaySound("misc/slide")
			"####" A 0 A_JumpIfInventory("SlideCooldown",1,"FatiguedSlide")
			"####" A 0 A_GiveInventory("SlideFatigue",1)
			"####" A 0 A_SetSpeed(10)
