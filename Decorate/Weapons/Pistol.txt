//Burst Pea Shooter
ACTOR DeimosHandgun : DeimosWeapon replaces Pistol 
{
Weapon.SelectionOrder 2100
Inventory.PickupMessage "Acquired a Handgun."
Weapon.AmmoType1 "HandgunMagazine"
Weapon.AmmoType2 "LowCalBullets"
Weapon.AmmoUse 1
Weapon.AmmoGive2 30
Weapon.SlotNumber 2
Decal "BulletChip"
AttackSound "weapons/gatfire"
Tag "UAC Security-9B"
States
{
Spawn:
PIST A -1
Stop
Ready:
TNT1 A 0 A_JumpIfInventory("HandgunMagazine",1,1)
Goto SlideLock
PISG A 0 A_JumpIf(getplayerinput(input_buttons) & bt_speed, "Movement")
PISG A 0 A_SetSpeed(0.6)
PISG A 1 A_WeaponReady(WRF_ALLOWRELOAD)
PISG A 0 A_SetCrosshair(85)
Loop

SlideLock:
PISG A 0 A_JumpIf(getplayerinput(input_buttons) & bt_speed, "SLMovement") //yea
PISG A 0 A_SetSpeed(0.6)
PISG C 1 A_WeaponReady(WRF_NOFIRE | WRF_ALLOWRELOAD)
PISG A 0 A_SetCrosshair(85)
Loop
Select: //maybe add SL'd versions of Select/Deselect, although I doubt many would notice
PISG A 1 A_Raise(22)
Loop
Deselect:
PISG A 1 A_Lower(16)
TNT1 A 0 A_SetCrosshair(98)
Loop


Fire: 
TNT1 A 0 A_JumpIfInventory("HandgunMagazine",1,1)
Goto SLRecoil
TNT1 A 0 A_JumpIf(getcrouchfactor < 1.0, "CrouchedFire")
TNT1 A 0 A_WeaponOffSet(0,44)
PISG B 1 A_GunFlash 
PISG B 1 A_FireBullets(1, 1, 1, Random(16,18), "BulletPuff",FBF_NORANDOM | FBF_USEAMMO)
TNT1 A 0 A_AlertMonsters
TNT1 A 0 A_SetPitch(pitch-1.3)
PISG C 2 Offset(0,44) A_WeaponReady(WRF_NOSWITCH | WRF_NOBOB)
PISG C 2 Offset(0,40) A_WeaponReady(WRF_NOSWITCH | WRF_NOBOB)
TNT1 A 0 A_JumpIfInventory("HandgunMagazine",1,1)
Goto SLRecoil
PISG B 2 Offset(0,36) A_WeaponReady(WRF_NOSWITCH | WRF_NOBOB)
PISG B 2 Offset(0,34) A_WeaponReady(WRF_NOSWITCH | WRF_NOBOB)
PISG A 1 Offset(0,32) A_WeaponReady(WRF_NOSWITCH | WRF_NOBOB)
PISG A 1
PISG A 1 A_Refire
Goto Ready

SLRecoil:
PISG C 2 Offset(0,36) A_WeaponReady(WRF_NOSWITCH | WRF_NOBOB)
PISG C 2 Offset(0,34) A_WeaponReady(WRF_NOSWITCH | WRF_NOBOB)
PISG C 1 Offset(0,32) A_WeaponReady(WRF_NOSWITCH | WRF_NOBOB)
PISG C 1
PISG C 1
Goto SlideLock

AltFire: //BRRT BRRT
TNT1 A 0 A_JumpIfInventory("HandgunMagazine",1,1)
Goto SLRecoil
TNT1 A 0 A_JumpIf(getcrouchfactor < 1.0, "CrouchedAltFire")
TNT1 A 0 A_WeaponOffSet(0,44)
TNT1 A 0 A_AlertMonsters
PISG B 0 A_GunFlash 
TNT1 A 0 A_TakeInventory("HandgunMagazine",1)
PISG B 1 A_FireBullets(1, 1, 1, Random(16,18), "BulletPuff",FBF_NORANDOM | FBF_USEAMMO)
PISG B 0 A_SetPitch(pitch-0.8)
PISG B 0 A_GunFlash 
PISG C 1 Offset(0,40) 
TNT1 A 0 A_JumpIfInventory("HandgunMagazine",1,1)
Goto SLRecoil
TNT1 A 0 A_TakeInventory("HandgunMagazine",1)
PISG B 1 A_FireBullets(1, 1, 1, Random(16,18), "BulletPuff",FBF_NORANDOM | FBF_USEAMMO)
PISG B 0 A_SetPitch(pitch-1.4)
TNT1 A 0 A_AlertMonsters
PISG B 0 A_GunFlash 
PISG C 1 Offset(0,42) 
TNT1 A 0 A_JumpIfInventory("HandgunMagazine",1,1)
Goto SLRecoil
TNT1 A 0 A_TakeInventory("HandgunMagazine",1)
PISG B 1 A_FireBullets(1, 1, 1, Random(16,18), "BulletPuff",FBF_NORANDOM | FBF_USEAMMO)
TNT1 A 0 A_AlertMonsters
PISG B 1
TNT1 A 0 A_SetPitch(pitch-2.6)
TNT1 A 0 A_SetPitch(pitch-Random(1,2))
TNT1 A 0 A_SetAngle(angle+Random(-3,3))
PISG C 2 Offset(0,44) 
PISG C 2 Offset(0,40) 
TNT1 A 0 A_JumpIfInventory("HandgunMagazine",1,1)
Goto SLRecoil
PISG B 2 Offset(0,36) A_WeaponReady(WRF_NOSWITCH | WRF_NOBOB)
PISG B 2 Offset(0,34) A_WeaponReady(WRF_NOSWITCH | WRF_NOBOB)
PISG A 1 Offset(0,32) A_WeaponReady(WRF_NOSWITCH | WRF_NOBOB)
PISG A 1
PISG A 1 A_Refire
Goto Ready

CrouchedFire: 
TNT1 A 0 A_WeaponOffSet(0,44)
PISG B 1 A_GunFlash 
TNT1 A 0 A_AlertMonsters
PISG B 1 A_FireBullets(1, 1, 1, Random(16,18), "BulletPuff",FBF_NORANDOM | FBF_USEAMMO)
TNT1 A 0 A_SetPitch(pitch-0.4)
PISG C 2 Offset(0,44) A_WeaponReady(WRF_NOSWITCH | WRF_NOBOB)
PISG C 2 Offset(0,40) A_WeaponReady(WRF_NOSWITCH | WRF_NOBOB)
TNT1 A 0 A_JumpIfInventory("HandgunMagazine",1,1)
Goto SLRecoil
PISG B 2 Offset(0,36) A_WeaponReady(WRF_NOSWITCH | WRF_NOBOB)
PISG B 2 Offset(0,34) A_WeaponReady(WRF_NOSWITCH | WRF_NOBOB)
PISG A 1 Offset(0,32) A_WeaponReady(WRF_NOSWITCH | WRF_NOBOB)
PISG A 1
PISG A 1 A_Refire
Goto Ready
CrouchedAltFire: //BRRT BRRT BUT MORE STABLE
TNT1 A 0 A_WeaponOffSet(0,44)
PISG B 0 A_GunFlash 
TNT1 A 0 A_AlertMonsters
TNT1 A 0 A_TakeInventory("HandgunMagazine",1)
PISG B 1 A_FireBullets(1, 1, 1, Random(16,18), "BulletPuff",FBF_NORANDOM | FBF_USEAMMO)
PISG B 0 A_SetPitch(pitch-0.5)
PISG B 0 A_GunFlash 
PISG C 1 Offset(0,40) 
TNT1 A 0 A_JumpIfInventory("HandgunMagazine",1,1)
Goto SLRecoil
TNT1 A 0 A_TakeInventory("HandgunMagazine",1)
PISG B 1 A_FireBullets(1, 1, 1, Random(16,18), "BulletPuff",FBF_NORANDOM | FBF_USEAMMO)
PISG B 0 A_SetPitch(pitch-0.9)
TNT1 A 0 A_AlertMonsters
PISG B 0 A_GunFlash 
PISG C 1 Offset(0,42) 
TNT1 A 0 A_JumpIfInventory("HandgunMagazine",1,1)
Goto SLRecoil
TNT1 A 0 A_TakeInventory("HandgunMagazine",1)
PISG B 1 A_FireBullets(1, 1, 1, Random(16,18), "BulletPuff",FBF_NORANDOM | FBF_USEAMMO)
TNT1 A 0 A_AlertMonsters
PISG B 1
TNT1 A 0 A_SetPitch(pitch-2.6)
TNT1 A 0 A_SetAngle(angle+Random(-1,1))
PISG C 2 Offset(0,44) 
PISG C 2 Offset(0,40) 
TNT1 A 0 A_JumpIfInventory("HandgunMagazine",1,1)
Goto SLRecoil
PISG B 2 Offset(0,36) A_WeaponReady(WRF_NOSWITCH | WRF_NOBOB)
PISG B 2 Offset(0,34) A_WeaponReady(WRF_NOSWITCH | WRF_NOBOB)
PISG A 1 Offset(0,32) A_WeaponReady(WRF_NOSWITCH | WRF_NOBOB)
PISG A 1
PISG A 1 A_Refire
Goto Ready
Flash:
PISF A 0 BRIGHT OffSet(0,70)
PISF A 2 BRIGHT OffSet(0,70) A_Light2 
TNT1 A 1 A_Light0
Goto LightDone

Reload: 
  TNT1 A 0 A_SetCrosshair(98)
  TNT1 A 0 A_JumpIfInventory("HandgunMagazine",1,1)
  Goto SLReload
  PISG A 6
  TNT1 A 0 A_JumpIf(getplayerinput(input_buttons) & bt_reload, "CheckAmmo") 
  PISG A 0 A_JumpIfInventory("HandgunMagazine", 15, "Ready")
  PISG A 0 A_JumpIfInventory("LowCalBullets", 1, "ReloadWork")
  Goto Ready
 ReloadWork: 
  PISG A 0 
  "####" A 0 A_OverlayPivotAlign(PSP_WEAPON,PSPA_RIGHT,PSPA_BOTTOM)
  "####" A 1 A_WeaponOffset(-2,35,WOF_INTERPOLATE)
  PISG A 0 A_StartSound("weapons/hgmagout",CHAN_AUTO) 
  "####" A 0 A_OverlayRotate(PSP_WEAPON,-1,WOF_ADD)
  "####" A 1 A_WeaponOffset(-4,38,WOF_INTERPOLATE)
  "####" A 0 A_OverlayRotate(PSP_WEAPON,-2,WOF_ADD)
  "####" A 1 A_WeaponOffset(-6,44,WOF_INTERPOLATE)
  "####" A 10 
 ReloadLoop: 
  PISG A 0 A_JumpIfInventory("HandgunMagazine",15,"ReloadFinish")
  "####" A 0 A_JumpIfInventory("LowCalBullets",1,1)
  Goto ReloadFinish
  "####" A 0 A_TakeInventory("LowCalBullets", 1)
  "####" A 0 A_GiveInventory("HandgunMagazine", 1) 
  Loop
 ReloadFinish: 
  PISG A 1 A_WeaponOffset(-6,50,WOF_INTERPOLATE)
  "####" A 12 A_PlaySound("weapons/hgmagin",CHAN_AUTO) 
  "####" A 5
  "####" A 1 A_WeaponOffset(-4,42,WOF_INTERPOLATE)
  "####" A 0 A_OverlayRotate(PSP_WEAPON,1,WOF_ADD)
  "####" A 1 A_WeaponOffset(-2,40,WOF_INTERPOLATE)
  "####" A 0 A_OverlayRotate(PSP_WEAPON,1,WOF_ADD)
  "####" A 1 A_WeaponOffset(0,38,WOF_INTERPOLATE)
  "####" A 0 A_OverlayRotate(PSP_WEAPON,1,WOF_ADD)
  "####" A 1 A_WeaponOffset(0,36,WOF_INTERPOLATE)
  "####" A 1 A_WeaponOffset(0,34,WOF_INTERPOLATE)
  "####" A 1 A_WeaponOffset(0,32,WOF_INTERPOLATE)
  goto Ready
  
 SLReload:
  PISG C 6
  TNT1 A 0 A_JumpIf(getplayerinput(input_buttons) & bt_reload, "SLCheckAmmo") 
  PISG C 0 A_JumpIfInventory("LowCalBullets", 1, "SLReloadWork")
  Goto SlideLock
 SLReloadWork: 
  PISG C 0 
  "####" C 0 A_OverlayPivotAlign(PSP_WEAPON,PSPA_RIGHT,PSPA_BOTTOM)
  "####" C 1 A_WeaponOffset(-2,35,WOF_INTERPOLATE)
  PISG C 0 A_StartSound("weapons/hgmagout",CHAN_AUTO) 
  "####" C 0 A_OverlayRotate(PSP_WEAPON,-1,WOF_ADD)
  "####" C 1 A_WeaponOffset(-4,38,WOF_INTERPOLATE)
  "####" C 0 A_OverlayRotate(PSP_WEAPON,-2,WOF_ADD)
  "####" C 1 A_WeaponOffset(-6,44,WOF_INTERPOLATE)
  "####" C 10 
 SLReloadLoop: 
  PISG C 0 A_JumpIfInventory("HandgunMagazine",15,"SLReloadFinish")
  "####" C 0 A_JumpIfInventory("LowCalBullets",1,1)
  Goto SLReloadFinish
  "####" C 0 A_TakeInventory("LowCalBullets", 1)
  "####" C 0 A_GiveInventory("HandgunMagazine", 1) 
  Loop
 SLReloadFinish: 
  PISG C 1 A_WeaponOffset(-6,50,WOF_INTERPOLATE)
  "####" C 10 A_PlaySound("weapons/hgmagin",CHAN_AUTO) 
  "####" C 16
  "####" A 4 A_PlaySound("weapons/hgslide",CHAN_AUTO)
  "####" A 1 A_WeaponOffset(-4,42,WOF_INTERPOLATE)
  "####" A 0 A_OverlayRotate(PSP_WEAPON,1,WOF_ADD)
  "####" A 1 A_WeaponOffset(-2,40,WOF_INTERPOLATE)
  "####" A 0 A_OverlayRotate(PSP_WEAPON,1,WOF_ADD)
  "####" A 1 A_WeaponOffset(0,38,WOF_INTERPOLATE)
  "####" A 0 A_OverlayRotate(PSP_WEAPON,1,WOF_ADD)
  "####" A 1 A_WeaponOffset(0,36,WOF_INTERPOLATE)
  "####" A 1 A_WeaponOffset(0,34,WOF_INTERPOLATE)
  "####" A 1 A_WeaponOffset(0,32,WOF_INTERPOLATE)
  goto Ready
  
 
   
   CheckAmmo:
   PISG A 1 A_WeaponOffset(2,35,WOF_INTERPOLATE)
   "####" A 1 A_WeaponOffset(4,38,WOF_INTERPOLATE)
   "####" A 1 A_WeaponOffset(6,44,WOF_INTERPOLATE)
   "####" A 1 A_WeaponOffset(8,52,WOF_INTERPOLATE)
   "####" A 1 A_WeaponOffset(12,62,WOF_INTERPOLATE)
   "####" A 1 A_WeaponOffset(18,72,WOF_INTERPOLATE)
   "####" A 0 A_GiveInventory("AmmoCheck")
   CheckLoop:
   PISG A 1
   TNT1 A 0 A_JumpIf(getplayerinput(input_buttons) & bt_reload, "CheckLoop")
   TNT1 A 0 A_TakeInventory("AmmoCheck")
   CheckFinish:
   PISG A 1 A_WeaponOffset(18,72,WOF_INTERPOLATE)
   "####" A 1 A_WeaponOffset(12,62,WOF_INTERPOLATE)
   "####" A 1 A_WeaponOffset(8,52,WOF_INTERPOLATE)
   "####" A 1 A_WeaponOffset(6,44,WOF_INTERPOLATE)
   "####" A 1 A_WeaponOffset(4,38,WOF_INTERPOLATE)
   "####" A 1 A_WeaponOffset(2,35,WOF_INTERPOLATE)
   Goto Ready
   
   SLCheckAmmo:
   PISG C 1 A_WeaponOffset(2,35,WOF_INTERPOLATE)
   "####" C 1 A_WeaponOffset(4,38,WOF_INTERPOLATE)
   "####" C 1 A_WeaponOffset(6,44,WOF_INTERPOLATE)
   "####" C 1 A_WeaponOffset(8,52,WOF_INTERPOLATE)
   "####" C 1 A_WeaponOffset(12,62,WOF_INTERPOLATE)
   "####" C 1 A_WeaponOffset(18,72,WOF_INTERPOLATE)
   "####" C 0 A_GiveInventory("AmmoCheck")
   SLCheckLoop:
   PISG C 1
   PISG C 0 A_JumpIf(getplayerinput(input_buttons) & bt_reload, "SLCheckLoop")
   PISG C 0 A_TakeInventory("AmmoCheck")
   SLCheckFinish:
   PISG C 1 A_WeaponOffset(18,72,WOF_INTERPOLATE)
   "####" C 1 A_WeaponOffset(12,62,WOF_INTERPOLATE)
   "####" C 1 A_WeaponOffset(8,52,WOF_INTERPOLATE)
   "####" C 1 A_WeaponOffset(6,44,WOF_INTERPOLATE)
   "####" C 1 A_WeaponOffset(4,38,WOF_INTERPOLATE)
   "####" C 1 A_WeaponOffset(2,35,WOF_INTERPOLATE)
   Goto Ready

}
}

Actor HandgunMagazine : Ammo
{
		Inventory.Amount 1 
		Inventory.MaxAmount 15

		+INVENTORY.IGNORESKILL
