ACTOR CQC : DeimosWeapon Replaces Fist
{
  Weapon.SelectionOrder 3700
  Weapon.Kickback 100
  +WEAPON.WIMPY_WEAPON
  +WEAPON.MELEEWEAPON
  -WEAPON.NOAUTOFIRE
  Weapon.SlotNumber 1
  States
  {
  Ready:
    PUNG A 0 A_JumpIf(getplayerinput(input_buttons) & bt_speed, "Movement")
    PUNG A 0 A_SetSpeed(0.7)
	PUNG A 0 A_SetCrosshair(98)
    PUNG A 1 A_WeaponReady(WRF_ALLOWRELOAD)
    Loop
  Deselect:
    PUNG A 1 A_Lower(24)
    Loop
  Select:
    PUNG A 1 A_Raise(24)
    Loop
  Fire:
    PUNG B 1
    PUNG C 2 
	PUNG D 4 A_Punch
    PUNG D 2
    PUNG C 2
    PUNG B 2 
	PUNG B 2 A_Refire
	PUNG A 2 Offset(0,38)
	PUNG A 2 Offset(0,32)
    Goto Ready 
	
  SprintPunch: //or how to eff up demons without berserk 
    TNT1 A 0 A_JumpIfInventory("PowerStrength",1,"ZerkedSprintPunch")
    PUNG B 1 A_WeaponOffset(0,35,WOF_INTERPOLATE)
    PUNG C 2 A_SetSpeed(1.0)
	PUNG D 4 A_CustomPunch(Random(100,120),1,0,"RegPunchPuff",104,0,0,"BasicArmorBonus","*fist") 
    PUNG D 7
    PUNG C 2 A_SetSpeed(0.8)
    PUNG B 2 
	PUNG A 2 Offset(0,38)
	PUNG A 2 Offset(0,32)
    Goto Ready
	
  ZerkedSprintPunch:
	PUNG B 1 A_WeaponOffset(0,35,WOF_INTERPOLATE)
    PUNG C 2 A_SetSpeed(1.0)
	PUNG D 4 A_CustomPunch(350,1,0,"ZerkedPunchPuff",104,0,0,"BasicArmorBonus","*fist") 
    PUNG D 7
    PUNG C 2 A_SetSpeed(0.8)
    PUNG B 2 
	PUNG A 2 Offset(0,38)
	PUNG A 2 Offset(0,32)
    Goto Ready
	
  AltFire: 
	TNT1 A 0 A_JumpIfInventory("PowerStrength",1,"ZerkedPunch")
	PUNG A 1 A_WeaponOffset(0,35,WOF_INTERPOLATE)
	PUNG A 1 A_WeaponOffset(0,38,WOF_INTERPOLATE)
	PUNG A 1 A_WeaponOffset(0,40,WOF_INTERPOLATE)
	PUNG A 1 A_WeaponOffset(0,44,WOF_INTERPOLATE)
	PunchHold:
    PUNG A 1
	PUNG A 0 A_GiveInventory("PunchCharge",1)
	PUNG A 0 A_Refire("PunchHold")
	
	TNT1 A 0 A_JumpIfInventory("PunchCharge",15,"Charged")
	PUNG A 0 A_WeaponOffset(0,35,WOF_INTERPOLATE)
	PUNG B 1
    PUNG C 2 
	PUNG D 5 A_CustomPunch(Random(15,20),1,0,"RegPunchPuff",64,0,0,"BasicArmorBonus","*fist")
    PUNG D 2
    PUNG C 2
    PUNG B 2 
	PUNG A 2 Offset(0,38)
	PUNG A 2 Offset(0,32)
	PUNG A 0 A_TakeInventory("PunchCharge",15)
    Goto Ready
	
   Charged:
	PUNG A 0 A_WeaponOffset(0,35,WOF_INTERPOLATE)
	PUNG B 1 A_Recoil(-2)
    PUNG C 2 
	PUNG D 5 A_CustomPunch(80,1,0,"RegPunchPuff",64,0,0,"BasicArmorBonus","*fist")
    PUNG D 2
    PUNG C 2
    PUNG B 2 
	PUNG A 2 Offset(0,38)
	PUNG A 2 Offset(0,32)
	PUNG A 0 A_TakeInventory("PunchCharge",15)
    Goto Ready
	
  ZerkedPunch:
	PUNG A 1 A_WeaponOffset(0,35,WOF_INTERPOLATE)
	PUNG A 1 A_WeaponOffset(0,38,WOF_INTERPOLATE)
	PUNG A 1 A_WeaponOffset(0,40,WOF_INTERPOLATE)
	PUNG A 1 A_WeaponOffset(0,44,WOF_INTERPOLATE)
 ZerkedPunchHold:
    PUNG A 1
	PUNG A 0 A_GiveInventory("PunchCharge",1)
	PUNG A 0 A_Refire("ZerkedPunchHold")
	
	TNT1 A 0 A_JumpIfInventory("PunchCharge",15,"ZerkedCharged")
	PUNG A 0 A_WeaponOffset(0,35,WOF_INTERPOLATE)
	PUNG B 1
    PUNG C 2 
	PUNG D 5 A_CustomPunch(Random(15,20),1,0,"ZerkedPunchPuff",64,0,0,"BasicArmorBonus","*fist")
    PUNG D 2
    PUNG C 2
    PUNG B 2 
	PUNG A 2 Offset(0,38)
	PUNG A 2 Offset(0,32)
	PUNG A 0 A_TakeInventory("PunchCharge",15)
    Goto Ready
	
  ZerkedCharged:
	PUNG A 0 A_WeaponOffset(0,35,WOF_INTERPOLATE)
	PUNG B 1 A_Recoil(-2)
    PUNG C 2 
	PUNG D 5 A_CustomPunch(250,1,0,"ZerkedPunchPuff",64,0,0,"BasicArmorBonus","*fist")
    PUNG D 2
    PUNG C 2
    PUNG B 2 
	PUNG A 2 Offset(0,38)
	PUNG A 2 Offset(0,32)
	PUNG A 0 A_TakeInventory("PunchCharge",15)
    Goto Ready
	
 Reload: 
  TNT1 A 0 A_SetCrosshair(98)
  PUNG A 6
  TNT1 A 0 A_JumpIf(getplayerinput(input_buttons) & bt_reload, "CheckAmmo") 
  Goto Ready
  
  CheckAmmo:
   PUNG A 1 A_WeaponOffset(2,35,WOF_INTERPOLATE)
   "####" A 1 A_WeaponOffset(4,38,WOF_INTERPOLATE)
   "####" A 1 A_WeaponOffset(6,44,WOF_INTERPOLATE)
   "####" A 1 A_WeaponOffset(8,52,WOF_INTERPOLATE)
   "####" A 1 A_WeaponOffset(12,62,WOF_INTERPOLATE)
   "####" A 1 A_WeaponOffset(18,72,WOF_INTERPOLATE)
   "####" A 0 A_GiveInventory("AmmoCheck")
   CheckLoop:
   PUNG A 1
   TNT1 A 0 A_JumpIf(getplayerinput(input_buttons) & bt_reload, "CheckLoop")
   TNT1 A 0 A_TakeInventory("AmmoCheck")
   CheckFinish:
   PUNG A 1 A_WeaponOffset(18,72,WOF_INTERPOLATE)
   "####" A 1 A_WeaponOffset(12,62,WOF_INTERPOLATE)
   "####" A 1 A_WeaponOffset(8,52,WOF_INTERPOLATE)
   "####" A 1 A_WeaponOffset(6,44,WOF_INTERPOLATE)
   "####" A 1 A_WeaponOffset(4,38,WOF_INTERPOLATE)
   "####" A 1 A_WeaponOffset(2,35,WOF_INTERPOLATE)
   Goto Ready
  }
}
